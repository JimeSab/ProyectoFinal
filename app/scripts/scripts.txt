-- Especialidades
SELECT id, nombre, activo
FROM especialidades
ORDER BY id;

-- Servicios
SELECT id, nombre, precioBase, duracionMinutos, especialidadId, creadoEn, actualizadoEn
FROM servicios
ORDER BY id;

-- Servicios adicionales
SELECT id, nombre, precio, activo, creadoEn, actualizadoEn
FROM servicios_adicionales
ORDER BY id;

-- Empleados
SELECT e.id, e.codigoEmpleado, u.nombre, u.primerApellido, es.nombre AS especialidad
FROM empleados e
JOIN usuarios u ON e.usuarioId = u.id
JOIN especialidades es ON e.especialidadId = es.id
ORDER BY e.id;

-- Servicios por empleado
SELECT e.codigoEmpleado, u.nombre AS empleado, s.nombre AS servicio
FROM _empleadoServicios es
JOIN empleados e ON es.A = e.id
JOIN usuarios u ON e.usuarioId = u.id
JOIN servicios s ON es.B = s.id
ORDER BY e.id, s.id;

-- Restricciones
SELECT r.id, t.nombre AS tipo, r.empleadoId, r.fecha, r.horaInicio, r.horaFin, r.todoElDia, r.motivo, r.activo
FROM restricciones_horario r
JOIN tipos_restriccion_horario t ON r.tipoRestriccionId = t.id
ORDER BY r.fecha;

-- Citas
SELECT
    c.id,
    c.fecha,
    c.horaInicio,
    c.horaFin,
    u.nombre AS cliente,
    e.codigoEmpleado,
    s.nombre AS servicio,
    ec.nombre AS estado,
    c.precioServicio,
    c.costoTotal
FROM citas c
JOIN usuarios u ON c.clienteId = u.id
JOIN empleados e ON c.empleadoId = e.id
JOIN servicios s ON c.servicioId = s.id
JOIN estados_cita ec ON c.estadoCitaId = ec.id
ORDER BY c.fecha, c.horaInicio;

-- Cantidad de citas por estado
SELECT ec.nombre AS estado, COUNT(*) AS cantidad
FROM citas c
JOIN estados_cita ec ON c.estadoCitaId = ec.id
GROUP BY ec.id, ec.nombre
ORDER BY ec.id;
