-- 10.1. REASIGNAR UNA CITA PARA DEJAR UN SERVICIO SIN CITAS
-- ============================================================

-- Primero ejecutar: 
USE citas;

-- Luego ejecutar:

-- La cita de Pedicure SPA se reasigna a Esmaltado Regular.
-- Así se conserva la cantidad total de citas.

UPDATE citas
SET servicioId = (SELECT id FROM servicios WHERE nombre = 'Esmaltado Regular'),
    horaInicio = '13:00:00',
    horaFin = '14:00:00',
    duracionMinutos = 60,
    precioServicio = 7500.00,
    costoTotal = 7500.00 + costoAdicionales
WHERE servicioId = (SELECT id FROM servicios WHERE nombre = 'Pedicure SPA')
	AND empleadoId = (SELECT id FROM empleados WHERE codigoEmpleado = 'EMP-003')
    AND fecha = '2026-08-19'
    AND horaInicio = '13:00:00';

-- Luego ejecutar este: 
UPDATE citas
SET observaciones = 'Esmaltado regular.'
WHERE id = 6;
